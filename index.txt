  <style>
    * { box-sizing: border-box; }
    body { font-family: -apple-system, 'Apple SD Gothic Neo', 'Malgun Gothic', sans-serif;
           background:#f3f4f8; margin:0; padding:24px 16px; color:#222; }
    .card { max-width:420px; margin:0 auto; background:#fff; border-radius:18px;
            padding:28px 22px; box-shadow:0 4px 18px rgba(0,0,0,.08); text-align:center; }
    h1 { font-size:22px; margin:0 0 6px; }
    .sub { color:#777; font-size:14px; margin-bottom:22px; white-space:pre-line; }
    input, button { width:100%; font-size:18px; padding:14px; border-radius:12px; }
    input { border:1px solid #ccc; margin-bottom:12px; background:#fff; text-align:center; }
    button { border:none; background:#e8453c; color:#fff; font-weight:700; cursor:pointer; }
    button:disabled { background:#bbb; }
    .big { font-size:42px; font-weight:800; margin:10px 0; color:#e8453c; }
    .name { font-size:20px; font-weight:700; }
    .msg { margin-top:16px; font-size:16px; white-space:pre-line; }
    .warn { color:#c62828; }

<script>
  // ▼ Apps Script 웹앱 주소 (/exec)
  const API_URL = 'https://script.google.com/macros/s/AKfycbwSSFXmhWod1X-jgDKdWZW9tHflR8_LvWgjWQE18mmaoZGKb0YyIBoJKjHV5foxAdhoug/exec';

  // 기기 ID (브라우저에 저장)
  function getDeviceId() {
    let id = null;
    try { id = localStorage.getItem('bcDeviceId'); } catch (e) {}
    if (!id) {
      id = (window.crypto && crypto.randomUUID ? crypto.randomUUID()
            : Date.now().toString(36) + Math.random().toString(36).slice(2));
      try { localStorage.setItem('bcDeviceId', id); } catch (e) {}
    }
    return id;
  }
  const DEVICE_ID = getDeviceId();
  const CODE = new URLSearchParams(location.search).get('code') || '';
  const app = document.getElementById('app');

  function esc(s) { return String(s).replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c])); }
  function fail(e) { app.innerHTML = '<div class="msg warn">오류가 발생했습니다.<br>' + esc(e && e.message ? e.message : e) + '<br>잠시 후 QR을 다시 찍어주세요.</div>'; }
  function badCode(r) { app.innerHTML = '<h1>📻 방송부 출석</h1><div class="msg warn">' + esc(r.msg) + '</div>'; }
  const PIN_ATTR = 'type="password" inputmode="numeric" pattern="[0-9]*" maxlength="4" autocomplete="off"';

  // 서버 호출 (구글 로그인 쿠키를 보내지 않음)
  async function api(action, args) {
    const res = await fetch(API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ action: action, args: args })
    });
    const data = await res.json();
    if (data && data.error) throw new Error(data.error);
    return data;
  }

  function load() {
    if (!CODE) return badCode({ msg: '방송부 컴퓨터 화면의 QR을 찍어서 들어와주세요.' });
    api('getStatus', [DEVICE_ID, CODE]).then(render).catch(fail);
  }

  function render(s) {
    if (s.mode === 'BADCODE') return badCode(s);
    if (s.mode === 'REGISTERED') return showCheckIn(s.name, s.todayTime, s.needPin);

    // 등록 / 재연결 화면
    app.innerHTML =
      '<h1>📻 ' + (s.open ? '기기 등록' : '기기 다시 연결') + '</h1>' +
      '<div class="sub">' + (s.open
        ? '등록 기간: ' + esc(s.period) + '\n처음이면 비밀번호 4자리를 새로 정하세요.\n이미 등록했는데 이 화면이 보이면 그때 정한 비밀번호를 입력하세요.'
        : '이미 등록한 학생은 등록할 때 정한\n비밀번호 4자리를 입력하면 다시 연결돼요.') + '</div>' +
      '<input id="name" type="text" placeholder="이름" maxlength="10" autocomplete="off">' +
      '<input id="pin" ' + PIN_ATTR + ' placeholder="비밀번호 (숫자 4자리)">' +
      '<button id="regBtn">' + (s.open ? '등록하기' : '다시 연결하기') + '</button>' +
      '<div class="msg" id="msg"></div>' +
      '<div class="note">※ 비밀번호는 다른 사람에게 알려주지 마세요.<br>※ 비밀번호를 잊었으면 선생님께 문의하세요.</div>';

    document.getElementById('regBtn').onclick = function () {
      const name = document.getElementById('name').value.replace(/\s+/g, '');
      const pin = document.getElementById('pin').value.trim();
      if (!name) return alert('이름을 입력하세요.');
      if (!/^\d{4}$/.test(pin)) return alert('비밀번호는 숫자 4자리로 입력하세요.');
      if (!confirm(name + ' 학생으로 이 휴대폰을 연결할까요?')) return;
      const btn = this; btn.disabled = true;
      api('registerDevice', [DEVICE_ID, name, pin, CODE]).then(r => {
        if (r.mode === 'BADCODE') return badCode(r);
        if (r.ok) showCheckIn(r.name, null, false, r.relinked ? '이 휴대폰으로 다시 연결되었습니다.' : '');
        else { document.getElementById('msg').innerHTML = '<span class="warn">' + esc(r.msg) + '</span>'; btn.disabled = false; }
      }).catch(fail);
    };
  }

  function showCheckIn(name, todayTime, needPin, notice) {
    app.innerHTML =
      '<h1>📻 방송부 출석</h1>' +
      '<div class="name">' + esc(name) + '</div>' +
      '<div class="sub">' + new Date().toLocaleDateString('ko-KR', {month:'long', day:'numeric', weekday:'short'}) + '</div>' +
      (notice ? '<div class="msg ok" style="margin:-10px 0 16px">' + esc(notice) + '</div>' : '') +
      (todayTime ? doneHtml(todayTime, true) : '<button id="ciBtn">출석하기</button>') +
      '<div id="result"></div>' +
      (needPin
        ? '<div class="pinbox" id="pinbox"><p>🔒 비밀번호 4자리를 정해주세요.<br>휴대폰이 기기를 잊어버렸을 때 다시 연결하는 데 필요해요.</p>' +
          '<input id="newpin" ' + PIN_ATTR + ' placeholder="숫자 4자리">' +
          '<button id="pinBtn">비밀번호 저장</button><div class="msg" id="pinmsg"></div></div>'
        : '');

    const btn = document.getElementById('ciBtn');
    if (btn) btn.onclick = function () {
      btn.disabled = true; btn.textContent = '처리 중...';
      api('checkIn', [DEVICE_ID, CODE]).then(r => {
        if (r.mode === 'BADCODE') return badCode(r);
        if (!r.ok) { document.getElementById('result').innerHTML = '<div class="msg warn">' + esc(r.msg) + '</div>'; return; }
        btn.remove();
        document.getElementById('result').innerHTML = doneHtml(r.time, r.already);
      }).catch(fail);
    };

    const pinBtn = document.getElementById('pinBtn');
    if (pinBtn) pinBtn.onclick = function () {
      const pin = document.getElementById('newpin').value.trim();
      if (!/^\d{4}$/.test(pin)) return alert('숫자 4자리로 입력하세요.');
      pinBtn.disabled = true;
      api('setPin', [DEVICE_ID, pin, CODE]).then(r => {
        if (r.mode === 'BADCODE') return badCode(r);
        if (r.ok) document.getElementById('pinbox').innerHTML = '<div class="msg ok">🔒 비밀번호가 저장되었습니다.</div>';
        else { document.getElementById('pinmsg').innerHTML = '<span class="warn">' + esc(r.msg) + '</span>'; pinBtn.disabled = false; }
      }).catch(fail);
    };
  }

  function doneHtml(time, already) {
    return '<div class="msg">' + (already ? '오늘 이미 출석했어요' : '✅ 출석 완료!') + '</div>' +
           '<div class="big">' + esc(time) + '</div>';
  }

  load();
</script>
